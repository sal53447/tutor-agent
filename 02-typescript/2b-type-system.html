Part 2 · TypeScript · 2b

The Type System

// Primitives
let s: string; let n: number; let b: boolean;
let u: undefined; let nl: null; let big: bigint; let sym: symbol;

// Literals — very powerful
type Direction = "up" | "down" | "left" | "right";
type HTTPMethod = "GET" | "POST" | "PUT" | "DELETE";

// Unions
type ID = string | number;

// Intersections
type Named = { name: string };
type Aged  = { age: number };
type Person = Named & Aged;    // { name: string; age: number }

// Discriminated unions — the workhorse
type Shape =
  | { kind: "circle"; r: number }
  | { kind: "square"; side: number };

function area(s: Shape): number {
  switch (s.kind) {
    case "circle": return Math.PI * s.r ** 2;
    case "square": return s.side ** 2;
  }
}

// Tuples
type Point = [number, number];
type Pair = [T, U];

Narrowing — how TS tracks type flow

TypeScript performs control flow analysis: it narrows the type of a variable based on the checks you write. After a check, the type inside that branch is automatically more specific.

// typeof narrowing
function format(x: string | number) {
  if (typeof x === "string") return x.toUpperCase();  // x: string here
  return x.toFixed(2);                                // x: number here
}

// instanceof narrowing
function handle(err: Date | Error) {
  if (err instanceof Error) return err.message;       // err: Error
  return err.toISOString();                           // err: Date
}

// in narrowing — check for property presence
type Circle = { kind: "circle"; r: number };
type Rect   = { kind: "rect"; w: number; h: number };
function area2(s: Circle | Rect) {
  if ("r" in s) return Math.PI * s.r ** 2;           // s: Circle
  return s.w * s.h;                                   // s: Rect
}

// Truthiness narrowing
function greet(name: string | null) {
  if (!name) return "Hello, stranger";
  return `Hello, ${name}`;                            // name: string (null excluded)
}

// Discriminated union — narrowing via a shared literal field
// The switch on s.kind tells TS exactly which branch of Shape applies.
// Inside case "circle", s.r exists; TS errors if you access s.side there.
function area(s: Shape): number {
  switch (s.kind) {
    case "circle": return Math.PI * s.r ** 2;   // s: { kind:"circle"; r:number }
    case "square": return s.side ** 2;           // s: { kind:"square"; side:number }
  }
}
Why this matters

Narrowing is what makes unions useful rather than annoying. You don't need to cast — you check, and TS knows. If you find yourself writing as casts to access a union member, add a discriminant field instead.

any, unknown, and never

// any — disables type checking. Poisons anything it touches. Avoid.
function bad(x: any) { x.whatever();  // no error — TS gave up }

// unknown — safe top type. You must narrow before use.
function parse(raw: unknown) {
  if (typeof raw === "string") return raw.toUpperCase();  // ok after check
  throw new Error("not a string");
}

// never — the empty type. Arises in exhausted unions.
// If a value reaches type never, the code path is unreachable.
// Covered fully in 2f: exhaustive matching with never.
Rule

When a function can receive "anything", reach for unknown, not any. unknown forces you to narrow; any silently bypasses every check TS would otherwise perform.

Function types and type aliases

// Function type alias
type Handler = (req: Request, res: Response) => void;
type Predicate = (item: T) => boolean;
type Transform = (input: A) => B;

// Inline function type
function run(fn: (x: number) => string) { return fn(42); }

// Type aliases are compile-time only — erased at runtime.
// typeof SomeTypeAlias does NOT exist in JS output.
type UserId = string;   // gone at runtime — purely a TS construct

Interfaces vs type aliases

  • Type aliases (type) — can be any shape (unions, tuples, mapped types). Preferred default.
  • Interfaces — only object shapes. Can be extended via declaration merging (needed for augmenting library types). Slightly better perf in huge codebases.

Rule of thumb: interface for public API contracts, type for everything else. Or: pick one and be consistent.

Avoid

enum. It generates runtime code, has weird reverse-mapping behavior, and doesn't play well with tree-shaking. Use literal unions ("A" | "B" | "C") or as const objects instead. This is a TS team recommendation too.

// Instead of enum:
const Role = { Admin: "admin", User: "user" } as const;
type Role = typeof Role[keyof typeof Role];  // "admin" | "user"
