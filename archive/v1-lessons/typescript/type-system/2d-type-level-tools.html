Type-Level Tools

TS has a small type-level programming language. Most projects need a handful of patterns; only library authors go deep.

// keyof — get the keys of a type as a union
type UserKeys = keyof { id: string; name: string };  // "id" | "name"

// typeof — get the TYPE of a VALUE
const config = { port: 3000, host: "localhost" };
type Config = typeof config;

// Mapped types — transform all keys
type Readonly<T> = { readonly [K in keyof T]: T[K] };
type Partial<T>  = { [K in keyof T]?: T[K] };

// Conditional types + infer — explained below
type ReturnType<F> = F extends (...args: any[]) => infer R ? R : never;
type Async<T> = T extends Promise<infer U> ? U : T;

// Template literals
type Route = `/${string}`;
type Event<T extends string> = `on${Capitalize<T>}`;   // onClick, onHover...

The three utility types you'll use daily

type User = { id: string; name: string; email: string; role: "admin" | "user" };

// Pick — keep only the named keys
type UserPreview = Pick<User, "id" | "name">;
// { id: string; name: string }

// Omit — drop the named keys
type CreateUserDto = Omit<User, "id">;
// { name: string; email: string; role: "admin" | "user" }

// Record — typed dictionary
type RoleLabel = Record<"admin" | "user", string>;
// { admin: string; user: string }
const labels: RoleLabel = { admin: "Administrator", user: "Standard user" };

// Practical: combine Omit + Partial for update payloads
type UpdateUserDto = Partial<Omit<User, "id">>;
// all fields except id become optional

The infer keyword

infer lets you extract a type from another type inside a conditional. Read it as: "if this matches, name the captured part R and use it."

// Extract the element type of any array
type ElementOf<T> = T extends (infer E)[] ? E : never;
type A = ElementOf<string[]>;    // string
type B = ElementOf<number[][]>;  // number[]

// Extract the resolved value of a Promise (same as built-in Awaited<T>)
type Unwrap<T> = T extends Promise<infer U> ? U : T;
type C = Unwrap<Promise<User>>;  // User
type D = Unwrap<string>;         // string (not a Promise, falls to : T)

// Exclude / Extract
type Status = "pending" | "active" | "banned";
type OpenStatus  = Exclude<Status, "banned">;   // "pending" | "active"
type ClosedStatus = Extract<Status, "banned">;   // "banned"

// Parameters — useful for wrapping functions you don't own
function createUser(name: string, role: "admin" | "user") { /* ... */ }
type CreateArgs = Parameters<typeof createUser>;   // [string, "admin" | "user"]

satisfies — validate shape without widening

satisfies checks that a value matches a type but keeps the literal / narrower type for subsequent use. Moved here from idioms because it's a type-level operator — it evaluates entirely at compile time. (See 2f for the assertion functions that complement it.)

type Config = Record<string, string | number>;

// Without satisfies: port is widened to string | number
const cfg1: Config = { host: "localhost", port: 3000 };
cfg1.port.toFixed();   // Error — might be string

// With satisfies: shape is checked AND literals are preserved
const cfg2 = { host: "localhost", port: 3000 } satisfies Config;
cfg2.port.toFixed();   // OK — TS knows port: number here
cfg2.host.toUpperCase(); // OK — TS knows host: string here

// Catches wrong value type while keeping inference
const routes = {
  home: "/",
  about: "/about",
  blog: 42,   // Error: number is not assignable to `/${string}`
} satisfies Record<string, `/${string}`>;
