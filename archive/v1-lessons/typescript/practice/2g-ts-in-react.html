useRef typing

// DOM ref — initial value null, type is HTMLInputElement
// ref.current is HTMLInputElement | null (null until the element mounts)
const inputRef = useRef<HTMLInputElement>(null);
inputRef.current?.focus();

// Mutable value ref — not a DOM node, holds an arbitrary value
// Pass a non-null initial value; ref.current is number (never null)
const timerRef = useRef<number>(0);
timerRef.current = window.setTimeout(fn, 1000);
null vs 0 initial value

Pass null when attaching to a DOM element — React sets current after mount. Pass a real initial value (0, false, etc.) for mutable instance variables. The initial value determines whether current is nullable.

useReducer with discriminated union actions

This is the payoff of discriminated unions from 2b: each action is a separate type branch, and the reducer gets exhaustive checking for free.

type State = { count: number; status: "idle" | "loading" | "error" };

type Action =
  | { type: "increment" }
  | { type: "decrement" }
  | { type: "setStatus"; status: State["status"] };

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case "increment":  return { ...state, count: state.count + 1 };
    case "decrement":  return { ...state, count: state.count - 1 };
    case "setStatus":  return { ...state, status: action.status };
    // TS errors here if a new Action branch is added and not handled
  }
}

const [state, dispatch] = useReducer(reducer, { count: 0, status: "idle" });
dispatch({ type: "increment" });
dispatch({ type: "setStatus", status: "loading" });
// dispatch({ type: "typo" });   // Error — not a valid action

Context typing

type User = { id: string; name: string };
type UserContextValue = { user: User; logout: () => void };

// createContext requires a default — use null and handle it in the hook
const UserContext = React.createContext<UserContextValue | null>(null);

// Custom hook: throws if used outside the provider, returns a non-null value
function useUserContext(): UserContextValue {
  const ctx = React.useContext(UserContext);
  if (!ctx) throw new Error("useUserContext must be used within UserProvider");
  return ctx;
}

function UserProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  if (!user) return <LoginPage />;
  return (
    <UserContext.Provider value={{ user, logout: () => setUser(null) }}>
      {children}
    </UserContext.Provider>
  );
}

Extending native element props

// ComponentProps — grab all props a native element or component accepts
type NativeButtonProps = React.ComponentProps<"button">;

// Extend a native element, adding your own props
type ButtonProps = React.ComponentPropsWithoutRef<"button"> & {
  variant?: "primary" | "danger";
  loading?: boolean;
};

function Button({ variant = "primary", loading, children, ...rest }: ButtonProps) {
  // ...rest passes through className, disabled, onClick, aria-*, etc.
  return <button {...rest} data-variant={variant}>{children}</button>;
}

// Use ComponentPropsWithRef when you need to forward a ref
type InputProps = React.ComponentPropsWithRef<"input"> & { label: string };

Generic hooks

// A generic fetch hook — T flows from call site to return type
function useFetch<T>(url: string): { data: T | null; loading: boolean; error: Error | null } {
  const [data, setData]       = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState<Error | null>(null);

  useEffect(() => {
    fetch(url)
      .then(r => r.json() as Promise<T>)
      .then(setData)
      .catch(setError)
      .finally(() => setLoading(false));
  }, [url]);

  return { data, loading, error };
}

// Usage — T is inferred from the explicit type argument
const { data } = useFetch<User[]>("/api/users");
// data: User[] | null

Async event handlers

// Event handlers must return void, not Promise<void> — React doesn't await them.
// Type the handler as () => void even when the body is async.
function Form() {
  // Correct: the onClick prop type is MouseEventHandler, which expects void
  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>): Promise<void> => {
    e.preventDefault();
    await submitData();   // async work is fine, React just ignores the Promise
  };

  return <form onSubmit={handleSubmit}>...</form>;
}

// If you assign the handler to a typed variable, use Promise<void> as return type
// to suppress the "floating promise" lint rule — but the prop itself stays void.
Patterns to memorize
  • React.ReactNode for children (not JSX.Element — too narrow).
  • Don't type return values of components. Let TS infer.
  • Avoid React.FC. It adds implicit children and is going out of fashion.
  • For events, type the handler, not just the value — get autocomplete on e.target, e.currentTarget.
  • Use ComponentPropsWithoutRef when spreading native props; use ComponentPropsWithRef when forwarding a ref.
